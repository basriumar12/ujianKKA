Integrasi Otomatis ke Google Spreadsheet

Agar jawaban 30+ siswa otomatis masuk ke Google Spreadsheet Anda secara real-time, tempelkan URL Web App dari Google Apps Script di bawah ini:

Kode Google Apps Script v2 (Mendukung Tes Koneksi & Anti-Gagal) 
// 1. Fungsi Tes Koneksi saat URL dibuka atau diuji
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    status: "terhubung",
    pesan: "Google Apps Script KKA & IoT Aktif!"
  })).setMimeType(ContentService.MimeType.JSON);
}

// 2. Fungsi Menerima Jawaban & Skor Siswa ke Google Sheet
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000); // Antrean aman untuk 30+ siswa kirim bersamaan
  
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = {};
    
    if (e.parameter && e.parameter.payload) {
      data = JSON.parse(e.parameter.payload);
    } else if (e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    }
    
    // Buat header kolom otomatis jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Waktu Kirim",
        "Nama Siswa",
        "Kelas",
        "No Absen",
        "Skor PG (0-100)",
        "Jumlah Benar",
        "Jumlah Salah",
        "Total Soal PG",
        "Esai Terjawab",
        "Durasi",
        "Rincian Jawaban PG",
        "Rincian Jawaban Esai"
      ]);
      sheet.getRange(1, 1, 1, 12).setFontWeight("bold").setBackground("#e2e8f0");
    }
    
    // Format rincian PG & Esai agar mudah dibaca guru di sel Spreadsheet
    var ringkasanPG = "";
    if (data.jawabanPG) {
      for (var k in data.jawabanPG) {
        ringkasanPG += k + ": " + data.jawabanPG[k] + " | ";
      }
    }
    
    var ringkasanEsai = "";
    if (data.jawabanEsai) {
      for (var m in data.jawabanEsai) {
        if (data.jawabanEsai[m]) {
          ringkasanEsai += "[" + m + "] " + data.jawabanEsai[m] + "\n\n";
        }
      }
    }
    
    sheet.appendRow([
      new Date(),
      data.nama || "Tanpa Nama",
      data.kelas || "-",
      data.absen || "-",
      data.skorPG !== undefined ? data.skorPG : 0,
      data.jumlahBenar || 0,
      data.jumlahSalah || 0,
      data.totalPG || 0,
      data.esaiTerjawab || "0/0",
      data.durasi || "00:00",
      ringkasanPG || "-",
      ringkasanEsai || "-"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      nama: data.nama,
      skor: data.skorPG
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
