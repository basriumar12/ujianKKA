Integrasi Otomatis ke Google Spreadsheet

Agar jawaban 30+ siswa otomatis masuk ke Google Spreadsheet Anda secara real-time, tempelkan URL Web App dari Google Apps Script di bawah ini:

Kode Google Apps Script (Salin & Tempel di Extensions → Apps Script) 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  // Otomatis membuat judul kolom jika sheet masih kosong (tanpa perlu daftar nama siswa dulu)
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Waktu Kirim",
      "Nama Siswa",
      "Kelas",
      "No Absen",
      "Jumlah Benar PG",
      "Jumlah Salah PG",
      "Total Soal PG",
      "Nilai PG (0-100)",
      "Esai Terjawab",
      "Durasi Pengerjaan",
      "Detail Jawaban PG",
      "Detail Jawaban Esai"
    ]);
  }
  
  // Langsung mencatat nama siswa baru & hasil koreksi otomatis ke baris baru
  sheet.appendRow([
    new Date(),
    data.nama,
    data.kelas,
    data.absen,
    data.jumlahBenar,
    data.jumlahSalah,
    data.totalPG,
    data.skorPG,
    data.esaiTerjawab,
    data.durasi,
    JSON.stringify(data.jawabanPG),
    JSON.stringify(data.jawabanEsai)
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
